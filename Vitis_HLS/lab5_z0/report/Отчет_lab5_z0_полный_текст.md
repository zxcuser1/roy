1. Цель и исходные данные

Цель работы — исследовать влияние директив INLINE, DATAFLOW, STREAM, PIPELINE, UNROLL и 
ARRAY_RESHAPE на аппаратную структуру, задержку и использование ресурсов при синтезе C++-
функции. Дополнительно сравнивается оценка производительности HLS IP с измерениями однопоточной 
и OpenMP-реализаций на ПК.

Функция lab5_z0(A, B, RES) вызывает четыре подфункции. func1 формирует поэлементное произведение

разбиение на четыре функции: оно требуется для исследования распараллеливания и обмена 
промежуточными данными.

5.2. Программная реализация (C++ и OpenMP)

Для сравнения с HLS IP подготовлены две обычные программы на C++: однопоточная и многопоточная с 
OpenMP. Обе содержат те же четыре функции func1–func4 и вычисляют RES[i] = (A[i] + B[i])² − 
2·A[i]·B[i]. Для простоты используются массивы типа int и обычные циклы for; промежуточные массивы 
выделяются перед измерением времени.

В однопоточной версии каждая функция содержит обычный цикл. В OpenMP-версии перед циклом 
каждой подфункции добавляется #pragma omp parallel for. Прочая арифметика не меняется: это позволяет 
сравнить способы исполнения одного алгоритма.

5.2.1. Однопоточный вариант

Для примера ниже приведена функция func1. func2–func4 устроены аналогично: один цикл перебирает 
элементы массива.

void func1(const int A[], const int B[], int prod[], int sum[], int rows)
{
    for (int i = 0; i < rows; ++i) {
        prod[i] = A[i] * B[i];
        sum[i] = A[i] + B[i];
    }
}

Листинг 1 — Однопоточная обработка элементов (func1)

5.2.2. Многопоточный вариант (OpenMP)

В многопоточной версии данные массива разбиваются между потоками средствами OpenMP. Каждый 
поток обрабатывает свои индексы без записи в общие элементы выходного массива, поэтому 
дополнительная синхронизация внутри цикла не требуется.

void func1(const int A[], const int B[], int prod[], int sum[], int rows)
{
#pragma omp parallel for
    for (int i = 0; i < rows; ++i) {
        prod[i] = A[i] * B[i];
        sum[i] = A[i] + B[i];
    }
}

Листинг 2 — Параллельная обработка с OpenMP (func1)

5.2.3. Тестирование и измерение времени

Тест запускает функцию для пяти размеров ROWS. Перед измерением и после него массив RES 
сравнивается с независимым эталоном A[i]² + B[i]². Значения A и B ограничены диапазоном −500…500, 
что исключает переполнение 32-битного int при вычислениях.

Для каждого размера проводятся 33 вызова lab5_z0. Время измеряется только вокруг цикла вызовов, без 
инициализации входных данных, выделения памяти и проверки результата. В консоль выводится CSV: 
ROWS, режим, число потоков, 33 запуска, суммарное время, среднее время одного запуска и 
контрольная сумма.

auto start = std::chrono::steady_clock::now();
for (int k = 0; k < 33; ++k)
    lab5_z0(A, B, RES, rows, prod, sum, prod2, sum2);
auto stop = std::chrono::steady_clock::now();

интерфейсов. Поэтому это сравнение характеристик вычисления, а не полноценных систем с 
одинаковыми условиями ввода-вывода.

Для достоверного сравнения «от запуска до результата» необходимо измерить передачу данных и время 
работы IP на реальной плате, а аномальную точку OpenMP перемерить несколькими независимыми 
сериями.

А.4. Общий тест и измерение времени — lab5_z0_tb_sw.cpp

#include "lab5_z0_sw.h"
 
#include <chrono>
#include <cstring>
#include <iomanip>
#include <iostream>
 
#ifdef _OPENMP
#include <omp.h>
#endif
#ifdef _WIN32
#include <conio.h>
#endif
 
int main(int argc, char* argv[])
{
    bool no_wait = false;
    bool quick = false;
    for (int k = 1; k < argc; ++k) {

        if (std::strcmp(argv[k], "--no-wait") == 0) no_wait = true;
        else if (std::strcmp(argv[k], "--quick") == 0) quick = true;
        else {
            std::cerr << "Unknown argument: " << argv[k] << '\n';
            return 2;
        }
    }
 
    const int sizes[] = {8192, 32768, 131072, 524288, 2097152};
    const int count = quick ? 1 : 5;
    const int runs = quick ? 3 : 33;
 
#ifdef _OPENMP
    omp_set_dynamic(0);
    const char* mode = "openmp";
    int threads = omp_get_max_threads();
#else
    const char* mode = "single";
    int threads = 1;
#endif
 
    std::cout << "rows,mode,threads,runs,total_ms,avg_ms,checksum\n";
 
    for (int test = 0; test < count; ++test) {
        int rows = sizes[test];
        int* A = new int[rows];
        int* B = new int[rows];
        int* RES = new int[rows];
        int* prod = new int[rows];
        int* sum = new int[rows];
        int* prod2 = new int[rows];
        int* sum2 = new int[rows];
 
        for (int i = 0; i < rows; ++i) {
            // Диапазон ограничен, чтобы int не переполнялся при возведении в квадрат.
            A[i] = (i % 1001) - 500;
            B[i] = ((i * 7) % 1001) - 500;
        }
 
        // Проверяем корректность до измерений.
        lab5_z0(A, B, RES, rows, prod, sum, prod2, sum2);
        for (int i = 0; i < rows; ++i) {
            int expected = A[i] * A[i] + B[i] * B[i];
            if (RES[i] != expected) {
                std::cerr << "ERROR at i=" << i << '\n';
                delete[] A; delete[] B; delete[] RES;
                delete[] prod; delete[] sum; delete[] prod2; delete[] sum2;
                return 1;
            }
        }
 
        // Измеряем только 33 вызова функции, без выделения памяти и проверки.
        auto start = std::chrono::steady_clock::now();
        for (int k = 0; k < runs; ++k)
            lab5_z0(A, B, RES, rows, prod, sum, prod2, sum2);
        auto stop = std::chrono::steady_clock::now();
 
        long long checksum = 0;
        for (int i = 0; i < rows; ++i) {
            int expected = A[i] * A[i] + B[i] * B[i];
            if (RES[i] != expected) {
                std::cerr << "ERROR at i=" << i << '\n';
                delete[] A; delete[] B; delete[] RES;
                delete[] prod; delete[] sum; delete[] prod2; delete[] sum2;

                return 1;
            }
            checksum += RES[i];
        }
 
        double total_ms = std::chrono::duration<double, std::milli>(stop - start).count();
        std::cout << rows << ',' << mode << ',' << threads << ',' << runs << ','
                  << std::fixed << std::setprecision(6)
                  << total_ms << ',' << total_ms / runs << ',' << checksum << '\n';
 
        delete[] A; delete[] B; delete[] RES;
        delete[] prod; delete[] sum; delete[] prod2; delete[] sum2;
    }
 
    std::cout.flush();
    if (!no_wait) {
        std::cerr << "Press a key to exit...\n";
#ifdef _WIN32
        _getch();
#else
        std::cin.get(); // На Linux требуется Enter.
#endif
    }
    return 0;
}
